**For CIFAR-100 with L2P**
CUDA_VISIBLE_DEVICES=1 python -m torch.distributed.launch \
        --master_port='29503' \
        --use_env main.py \
        cifar100_l2p \
        --model vit_base_patch16_224 \
        --original_model vit_base_patch16_224 \
        --batch-size 16 \
        --data-path ../local_datasets/\
        --output_dir ./output \
        --seed 15 \
        --epochs 5
